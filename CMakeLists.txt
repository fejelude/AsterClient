cmake_minimum_required(VERSION 3.24)
project(AsterClient VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(ASTER_BOOTSTRAP_ONLY "Build only the unmodified upstream ImGui example" OFF)
option(ASTER_ASAN "Enable MSVC AddressSanitizer" OFF)
option(ASTER_CORE_ONLY "Build the core and test runner without the UI" OFF)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Aster requires Windows x64")
endif()
add_library(imgui STATIC external/imgui/imgui.cpp external/imgui/imgui_draw.cpp
  external/imgui/imgui_tables.cpp external/imgui/imgui_widgets.cpp
  external/imgui/imgui_demo.cpp external/imgui/backends/imgui_impl_win32.cpp
  external/imgui/backends/imgui_impl_dx11.cpp)
target_include_directories(imgui PUBLIC external/imgui external/imgui/backends)
target_compile_definitions(imgui PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(imgui PUBLIC d3d11 dxgi d3dcompiler dwmapi)
add_executable(ImGuiBaseline external/imgui/examples/example_win32_directx11/main.cpp)
target_link_libraries(ImGuiBaseline PRIVATE imgui)
if(ASTER_BOOTSTRAP_ONLY)
  return()
endif()
add_library(aster_core STATIC src/core/model.cpp src/core/planner.cpp src/config/store.cpp
  src/platform/windows.cpp src/auth/license.cpp src/diagnostics/log.cpp)
target_include_directories(aster_core PUBLIC src external/json)
target_compile_definitions(aster_core PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(aster_core PRIVATE /W4 /permissive- /utf-8 /EHsc)
target_link_libraries(aster_core PUBLIC user32 shell32 advapi32 crypt32 bcrypt winhttp version ole32 windowscodecs wintrust)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory(external/minhook EXCLUDE_FROM_ALL)
add_library(AsterBridge SHARED src/integration/bridge_dll.cpp)
target_include_directories(AsterBridge PRIVATE src external/bedrock-sdk external external/minhook/include)
target_compile_definitions(AsterBridge PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(AsterBridge PRIVATE /W4 /permissive- /utf-8 /EHa)
target_link_libraries(AsterBridge PRIVATE minhook)
target_link_options(AsterBridge PRIVATE /DYNAMICBASE /NXCOMPAT /Brepro)
add_executable(AsterBridgeTool tools/bridge_probe.cpp)
target_link_libraries(AsterBridgeTool PRIVATE aster_core)
target_compile_options(AsterBridgeTool PRIVATE /W4 /permissive- /utf-8 /EHsc)
enable_testing()
add_executable(AsterTests tests/tests.cpp)
target_link_libraries(AsterTests PRIVATE aster_core)
target_compile_options(AsterTests PRIVATE /W4 /permissive- /utf-8 /EHsc)
add_test(NAME CoreAndPlatform COMMAND AsterTests)
set_tests_properties(CoreAndPlatform PROPERTIES TIMEOUT 45)
if(ASTER_CORE_ONLY)
  return()
endif()
add_executable(AsterClient WIN32 src/app/main.cpp src/ui/interface.cpp src/app/app.rc src/app/app.manifest)
target_link_libraries(AsterClient PRIVATE aster_core imgui comdlg32)
target_compile_options(AsterClient PRIVATE /W4 /permissive- /utf-8 /EHsc)
target_link_options(AsterClient PRIVATE /DYNAMICBASE /NXCOMPAT /Brepro)
if(ASTER_ASAN)
  target_compile_options(aster_core PRIVATE /fsanitize=address /Zi)
  target_compile_options(AsterClient PRIVATE /fsanitize=address /Zi)
endif()
install(TARGETS AsterClient RUNTIME DESTINATION .)
install(FILES README.md LICENSE CHANGELOG.md dependencies.lock.json DESTINATION .)
install(DIRECTORY LICENSES/ DESTINATION LICENSES)
install(DIRECTORY docs/ DESTINATION docs PATTERN spec-review EXCLUDE)
